import pandas as pd
import numpy as np
import gc
import warnings
warnings.filterwarnings('ignore')

import lightgbm as lgb
from sklearn.model_selection import GroupKFold
from sklearn.metrics import roc_auc_score
INPUT_DIR = '/kaggle/input/datasets/pomykalovaanna/ieee-fraud-detection/'
OUTPUT_DIR = '/kaggle/working/'

train_trans = pd.read_csv(f'{INPUT_DIR}train_transaction.csv')
train_id = pd.read_csv(f'{INPUT_DIR}train_identity.csv')
test_trans = pd.read_csv(f'{INPUT_DIR}test_transaction.csv')
test_id = pd.read_csv(f'{INPUT_DIR}test_identity.csv')

test_id = test_id.rename(columns=lambda c: c.replace('id-', 'id_') if c.startswith('id-') else c)

train = train_trans.merge(train_id, on='TransactionID', how='left')
test = test_trans.merge(test_id, on='TransactionID', how='left')

y = train['isFraud'].copy()
test_ids = test['TransactionID'].copy()

train = train.drop(['isFraud', 'TransactionID'], axis=1)
test = test.drop(['TransactionID'], axis=1)

del train_trans, train_id, test_trans, test_id
gc.collect()

for df in [train, test]:
    df['hour'] = df['TransactionDT'] // 3600 % 24
    df['day_of_week'] = df['TransactionDT'] // (3600 * 24) % 7
    df['day'] = df['TransactionDT'] // (3600 * 24)

for df in [train, test]:
    df['uid'] = df['card1'].astype(str) + '_' + df['addr1'].astype(str) + '_' + \
                df['P_emaildomain'].astype(str) + '_' + df['day'].astype(str)

uid_agg_cols = ['TransactionAmt', 'C1', 'C13', 'D1', 'D4']
for col in uid_agg_cols:
    if col in train.columns:
        train[f'uid_{col}_mean'] = train.groupby('uid')[col].transform('mean')
        train[f'uid_{col}_std'] = train.groupby('uid')[col].transform('std')
        train[f'uid_{col}_count'] = train.groupby('uid')[col].transform('count')
        
        test[f'uid_{col}_mean'] = test.groupby('uid')[col].transform('mean')
        test[f'uid_{col}_std'] = test.groupby('uid')[col].transform('std')
        test[f'uid_{col}_count'] = test.groupby('uid')[col].transform('count')

c_cols = [c for c in train.columns if c.startswith('C') and len(c) <= 3]
for i in range(len(c_cols)):
    for j in range(i + 1, min(i + 5, len(c_cols))):
        c1, c2 = c_cols[i], c_cols[j]
        train[f'{c1}_mul_{c2}'] = train[c1] * train[c2]
        test[f'{c1}_mul_{c2}'] = test[c1] * test[c2]

train_cols, test_cols = set(train.columns), set(test.columns)
common_cols = [c for c in train.columns if c in test_cols]
train = train[common_cols].copy()
test = test[common_cols].copy()

cat_cols = [c for c in train.columns if train[c].dtype == 'object' or train[c].dtype.name == 'category']

for col in cat_cols:
    combined = pd.concat([train[col], test[col]], axis=0).astype('str')
    categories = pd.Categorical(combined).categories
    train[col] = pd.Categorical(train[col].astype('str'), categories=categories)
    test[col] = pd.Categorical(test[col].astype('str'), categories=categories)

num_cols = [c for c in train.columns if c not in cat_cols]
train[num_cols] = train[num_cols].fillna(-999)
test[num_cols] = test[num_cols].fillna(-999)

groups = train['uid'].astype('category').cat.codes
n_splits = 5
gkf = GroupKFold(n_splits=n_splits)

lgb_params = {
    'objective': 'binary',
    'metric': 'auc',
    'boosting_type': 'gbdt',
    'learning_rate': 0.05,
    'num_leaves': 256,
    'max_depth': -1,
    'feature_fraction': 0.8,
    'bagging_fraction': 0.8,
    'bagging_freq': 1,
    'lambda_l1': 0.1,
    'lambda_l2': 0.1,
    'verbose': -1,
    'n_jobs': -1,
    'seed': 42
}

oof_preds = np.zeros(len(train))
test_preds = np.zeros(len(test))

for fold, (train_idx, valid_idx) in enumerate(gkf.split(train, y, groups=groups)):
    X_train, X_valid = train.iloc[train_idx], train.iloc[valid_idx]
    y_train, y_valid = y.iloc[train_idx], y.iloc[valid_idx]
    
    dtrain = lgb.Dataset(X_train, label=y_train, categorical_feature=cat_cols)
    dvalid = lgb.Dataset(X_valid, label=y_valid, categorical_feature=cat_cols)
    
    model = lgb.train(
        lgb_params,
        dtrain,
        num_boost_round=2000,
        valid_sets=[dtrain, dvalid],
        callbacks=[lgb.early_stopping(stopping_rounds=100), lgb.log_evaluation(period=0)]
    )
    
    oof_preds[valid_idx] = model.predict(X_valid, num_iteration=model.best_iteration)
    test_preds += model.predict(test, num_iteration=model.best_iteration) / n_splits

submission = pd.DataFrame({
    'TransactionID': test_ids,
    'isFraud': test_preds
})
submission.to_csv(f'{OUTPUT_DIR}submission_ieee.csv', index=False)
